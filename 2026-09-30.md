from openpyxl import load_workbook

wb = load_workbook("questions.xlsx", read_only=True)
ws = wb.active

questions = [
    row[0].strip()
    for row in ws.iter_rows(
        min_row=2,
        min_col=2,
        max_col=2,
        values_only=True,
    )
    if isinstance(row[0], str) and row[0].strip()
]

wb.close()
